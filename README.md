A = [[1, 2, 3],
     [3, 6, 9]]

if A[0][0] * A[1][1] == A[0][1] * A[1][0] and A[0][0] * A[1][2] == A[0][2] * A[1][0]:
    rank = 1
else:
    rank = 2

print(rank)
